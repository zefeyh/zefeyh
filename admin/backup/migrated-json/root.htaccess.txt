# 网址导航 —— 服务器安全与性能规则（Apache；Nginx 请在 server 配置中等效实现）
Options -Indexes

# 1. 敏感文件禁止直接访问（后台仍可通过 PHP 文件系统读取）
<FilesMatch "^(admin-config\.php|nav-template\.html|\.login-throttle\.json|.*\.(bak|old|log|sql|swp|tmp))$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>

# 2. 备份目录禁止访问（admin.php 会自动写入该目录的 .htaccess）
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteRule ^backup(/|$) - [F,L]
    # 目录内禁止执行任何脚本（防止上传目录出现 WebShell）
    RewriteRule ^ico/.*\.(php|phtml|phar|pht|cgi|pl|py|asp|jsp|sh)([0-9]*)?$ - [F,L,NC]
</IfModule>

# 3. 安全响应头（后台 PHP 自行发送专属头，这里只作用于静态文件，避免重复头）
<IfModule mod_headers.c>
    <FilesMatch "\.(html?|css|js|json|xml|ico|png|jpe?g|gif|svg|webp|woff2?|ttf)$">
        Header set X-Content-Type-Options "nosniff"
        Header set X-Frame-Options "DENY"
        Header set Referrer-Policy "no-referrer"
        Header set Permissions-Policy "geolocation=(), microphone=(), camera=(), payment=()"
        Header set Cross-Origin-Resource-Policy "same-origin"
    </FilesMatch>
</IfModule>

# 4. 缓存策略：静态资源长缓存，HTML 短缓存便于生成后立即生效
<IfModule mod_headers.c>
    <FilesMatch "^(font-awesome\.min\.css|.*\.(ico|png|jpe?g|gif|svg|webp|woff2?|ttf|js))$">
        Header set Cache-Control "public, max-age=2592000, immutable"
    </FilesMatch>
    # 后台样式表可能被修改，短缓存便于刷新后立即生效
    <FilesMatch "^admin\.css$">
        Header set Cache-Control "public, max-age=600, must-revalidate"
    </FilesMatch>
    <FilesMatch "\.json$">
        Header set Cache-Control "public, max-age=60, must-revalidate"
    </FilesMatch>
    <FilesMatch "\.html$">
        Header set Cache-Control "public, max-age=60, must-revalidate"
    </FilesMatch>
    <FilesMatch "admin\.php$">
        Header set Cache-Control "private, no-store, no-cache, must-revalidate"
    </FilesMatch>
</IfModule>

# 5. 压缩文本资源，减少传输体积
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/css text/plain application/json application/javascript
</IfModule>
